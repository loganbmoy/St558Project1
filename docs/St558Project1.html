Introduction

For this assignment we will investigate the data within the PUMS Census API. PUMS includes information about America’s economy and population. We’ll examine data from years 2021-2024 and look at the variables AGEP, GASP, GRPIP, JWAP, JWDP, JWMNP, PWGTP, FER, HHL, SCH, SCHL, SEX, ST/STATE , REGION, and DIVISION.

First we need some functions to actually grab the data from that API. Since the data is in a JSON format we’ll need to use functions in httr and jsonlite package to obtain the data.

Since the API gives us information about the variables in it, we’ll read those in as well. This helps us get information like factor names for categorical variables.

getVars <- function(url) {
  
  varInfo <- GET(url)
  parsed <- fromJSON(rawToChar(varInfo$content))
  vars <- parsed$variables
  vars <- vars[names(vars) %in% varNames]
  return (vars)
}

Making a function to parse API data

Here we’ll start reading in the data. Process is broken up into multiple helper functions.

We read in variable name data we stored earlier.

We make sure the geography levels and values are legal.

We have a function that creates url information that makes an url that lets us request the year, geography, and variables we want.

This url is then used by another function which turns the data from that url into a tibble.

Then we make sure the tibble’s columns are in the correct format.

All of this gets wrapped up in a wrapper function that gives us that year’s information.

# Helper function that makes sure we have correct geo variables
# @param int year the year we are interested in
# @param geoLvl string the geography level we want
# @param geoVal string the value of the geography level
# @returns a list containing geoLvl and geoVal
geoValidator <- function(year, geoLvl, geoVal) {
  
  # Only one geography level allowed
  if (length(geoLvl) != 1) {
    stop("Only one geography level allowed.")
  }
  
  # Check if the geoLvl is valid
  legal <- c("REGION", "DIVISION", "STATE")
  if (!(toupper(geoLvl) %in% legal)) {
    stop("Invalid geoLvl given. Valid options are Region, Division, or State.")
  }

  yearVar <- varData[[year - 2020]]

  # Making sure geoLvl is caps and deals with Year 2021 and 2022 case
  geoLvl <- switch(
    geoLvl,
    REGION   = "REGION",
    DIVISION = "DIVISION",
    STATE    =  if_else(year < 2023, "ST", "STATE") 
  )

  # Obtain valid values from RDS, names are numbers and its value is the label
  validVals <- names(yearVar[[geoLvl]]$values$item)

  # If we use default geoVal then use the first of the valid values
  if (is.null(geoVal)) {
    geoVal <- validVals[1]
  }

  # All option
  if (toupper(geoVal) == "ALL") {
    return (list(geoLvl = geoLvl, geoVal = NULL))
  }

  # Make sure the geoVal is actually a valid option
  if (!(geoVal %in% validVals)) {
    stop(paste("Invalid value for", geoLvl, ". Must be one of:", paste(validVals, collapse = ", ")))
  }
  return (list(geoLvl = geoLvl, geoVal = geoVal))
}

# Helper function that takes url and gives us the content as a tibble
getUrlInfo <- function(urlString) {
  
  dataInfo <- GET(urlString)
  parsed <- fromJSON(rawToChar(dataInfo$content))
  
  # Col names are the first to appear, need to use subsetting to get them
  parsedNames <- parsed[1, ]
  parsed <- parsed[-1, , drop = FALSE]
  colnames(parsed) <- parsedNames
  
  parsed <- as_tibble(parsed)
  
  return (parsed)
}

# Helper for converting Census time codes to midpoint times
parse_time_midpoint <- function(x) {
  
  x <- as.character(x)
  
  sapply(x, function(val) {
    # Missing values
    if (is.na(val) || val %in% c("b", "bb", "bbbb", "N.A.", "")) {
      return(NA)
    }
    
    val <- sprintf("%04d", as.integer(val))
    hour <- as.numeric(substr(val, 1, 2))
    minute <- as.numeric(substr(val, 3, 4))
    hm(sprintf("%02d:%02d", hour, minute))
  })
}

# Helper function for fixing the numerical variables
# @param pums tibble we selecting vars from
# @param varNumeric string vector of numeric vars we want
# @return pums tibble with the correct numerical variables
# @throws error when varNumeric does not contain numeric vars
fixNum <- function(pums, varNumeric) {
  
  time_vars <- c("JWAP", "JWDP")
  
  # Loop that makes varNumeric variables in pums numeric, also does time midpoint
  for (v in varNumeric) {
    if (v %in% time_vars) {
      pums[[v]] <- parse_time_midpoint(pums[[v]])

    } else {
      pums[[v]] <- as.numeric(pums[[v]])
    }
  }
  return (pums)
}

# Helper function for fixing the categorical variables
# @param pums tibble we selecting vars from
# @param year int we selecting vars for
# @param varCategorical string vector of categorical vars we want
# @return pums tibble with the correct categorical variables
# @throws error when varCategorical does not contain categorical vars
fixCate <- function(pums, year, varCategorical) {
  
  yearVar <- varData[[year - 2020]]
  
  # Loop that makes varCategorical variables in pums categorical
  for (v in varCategorical) {
    # values and items takes you to factor levels (numbers)
    pums[[v]] <- factor(pums[[v]], levels = names(yearVar[[v]]$values$item), labels = unlist(yearVar[[v]]$values$item))
  }
  return (pums)
}


# Parses the data, uses above helper functions.
# Gives us a tibble of data from Census API. 
# Subsets the data based on geography, year, and vars
# @param geoLvl string the geography level we want, default is STATE
# @param geoVal string the value of the geography level
# @param year int the year, default 2024
# @param varNumeric string a vector of numeric variables to return, default c("AGEP", "PWGTP") 
# @param varCategorical string a vector of categorical variables to return, default c("SEX")
# @returns a tibble with the specified information
# @throws error when year is not 2021-2024
# @throws error when varNumeric does not contain PWGTP or length < 2
# @throws error when varCategorical length == 0
getYearData <- function(geoLvl = "STATE", geoVal = NULL, year = 2024, varNumeric = c("AGEP", "PWGTP"), varCategorical = c("SEX") ) {

  # Check if the year is valid
  if (!(year %in% c(2021, 2022, 2023, 2024))) {
    stop("Invalid year given. Year must be numeric and between 2021-2024 (inclusive).")
  }
  
  # Check if the varNumeric is valid
  if ((!("PWGTP" %in% varNumeric)) | (length(varNumeric) < 2)) {
    stop("Invalid numeric variables given. Vector must be made of strings. Vector must contain PWGTP and have length >= 2.")
  }
  
  # Check if varNumeric contains legal variables
  legalN <- c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP", "PWGTP")
  if (!(all(varNumeric %in% legalN))) {
    stop("varNumeric contains variables that aren't numeric.")
  }
  
  # Check if the varCategorical is valid
  if (length(varCategorical)  == 0) {
    stop("Vector must be made of strings. Vector must have length > 0.")
  }
  
  # Check if varCategorical contains legal variables
  legalC <- c("FER", "HHL", "SCH", "SCHL", "SEX")
  if (!(all(varCategorical %in% legalC))) {
    stop("varCategorical contains variables that aren't categorical.")
  }
  
  # Validates the geo variables
  validGeo <- geoValidator(year, geoLvl, geoVal)
  
  # Creating URL
  urlString <- createURL(validGeo[[1]], validGeo[[2]], year, varNumeric, varCategorical)
  
  # Getting the tibble
  pums <- getUrlInfo(urlString)
  
  # Numeric variables
  pums <- fixNum(pums, varNumeric)
  
  # Categorical variables
  pums <- fixCate(pums, year, varCategorical)
  
  class(pums) <- c("census", class(pums))
  
  return (pums)
}

Combining Years function

Here is the final form of the function we use to get data from the API. This function uses the previous getYearData and a loop to get data from multiple years. This allows the user to compare two+ years (although this function can still take 1 year).

# Allows you to get multiple years worth of data from Census API 
# @param year can be a vector of years
# rest comes from getYearData
getCensusData <- function(geoLvl = "STATE", geoVal = NULL, year = 2024, varNumeric = c("AGEP", "PWGTP"), varCategorical = c("SEX")) {

  # Check years
  if (!all(year %in% c(2021, 2022, 2023, 2024))) {
    stop("Invalid year given. Year must be numeric and between 2021-2024 (inclusive).")
  }
  
  # Better to use some type of apply function, but not covered yet
  pums <- getYearData(geoLvl, geoVal, year[1], varNumeric, varCategorical)
  pums$YEAR <- year[1]
  dataList <- list(pums)
  
  # Starting from 2nd year in vector of years, make sure there is more than 1 year first
  if (length(year) >= 2) {
    for (i in 2:length(year)) {
      pums <- getYearData(geoLvl, geoVal, year[i], varNumeric, varCategorical)
      pums$YEAR <- year[i]
      dataList[[length(dataList) + 1]] <- pums
    }
  }

  # Combine all years
  final <- bind_rows(dataList)
  final$YEAR <- factor(final$YEAR)
  class(final) <- c("census", class(final))

  return (final)
}

Function for summarizing

# Helper summary function for our census data (numeric)
# Calculates the mean and sd. Uses PWGTP as a weight vector
# Does not look at time variables
# @param census a census tibble 
# @param varNumeric string of numeric variables we want
# @returns a list with means and sd of numeric 
sumCenNum <- function(census, varNumeric) {

  # Vectors to store means and sds, has same name
  means <- numeric(length(varNumeric))
  sds <- numeric(length(varNumeric))
  names(means) <- varNumeric
  names(sds) <- varNumeric

  # Calculating the summary stats
  for (v in varNumeric) {
    mean <- sum(census[[v]] * census$PWGTP) / sum(census$PWGTP)
    sd <- sqrt(sum(census[[v]]^2 * census$PWGTP) / sum(census$PWGTP) - mean^2)

    means[v] <- mean
    sds[v] <- sd
  }

  return (list(means = means,sd = sds))
}

# Helper summary function for our census data (categorical)
# @param census a census tibble 
# @param varCategorical string of numeric variables we want
# @returns a list with counts
sumCenCat <- function(census, varCategorical) {
  
  counts <- list()

  for (v in varCategorical) {
    lvl <- levels(census[[v]])
    countLvl <- numeric(length(lvl))
    names(countLvl) <- lvl
    # Summing for every level of a factor based on weight
    for (l in lvl) {
      countLvl[l] <- sum(census$PWGTP[census[[v]] == l], na.rm = TRUE)
    }
    counts[[v]] <- countLvl
  }
  return (counts)

}

# Summary function for our census data
# @param census a census tibble coming from getCensusData() (as it has year variable)
# @param varNumeric string of numeric variables we want, default is all
# @param varCategorical string of categorical variables we want, default is all
# @returns a list with means and sd of numeric and counts of categorical
# @throws error when vector of variables contains variables not in census or when using
#   weight or time variables
summary.census <- function(census, varNumeric = NULL, varCategorical = NULL) {
  
  # Check if varNumeric contains legal variables
  legalN <- c("AGEP", "GASP", "GRPIP", "JWMNP")
  
  if (is.null(varNumeric)) {
    varNumeric <- intersect(legalN, names(census))
  }
  
  if (!(all(varNumeric %in% legalN))) {
    stop("varNumeric contains variables that aren't numeric. Or it contains time or weight variables.")
  }
  
  # Check if varCategorical contains legal variables
  legalC <- c("FER", "HHL", "SCH", "SCHL", "SEX", "YEAR")
  
  if (is.null(varCategorical)) {
    varCategorical <- intersect(legalC, names(census))
  }
  
  if (!(all(varCategorical %in% legalC))) {
    stop("varCategorical contains variables that aren't categorical.")
  }
  
  summaryC <- sumCenNum(census, varNumeric)
  summaryC$count <- sumCenCat(census, varCategorical)
  
  return (summaryC)
}
